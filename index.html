// Sheet 1 (OverTime Details) & Sheet 2 (Payroll & Allowances) URLs const SHEET_1_URL = 'https://docs.google.com/spreadsheets/d/1wL36ZedH-wVc7NVTiBzwPGTxYy-8fPCEnUebPmrMZQ0/gviz/tq?tqx=out:csv&gid=552193494'; const SHEET_2_URL = 'https://docs.google.com/spreadsheets/d/1EZm6Kyrp6h4rUBuUv0l7HHd3IRQvqT2-10xs4YzyML0/gviz/tq?tqx=out:csv&gid=0'; // CSV Parsing Function (Header Spaces और Quotes को Clean करता है) function parseCSV(csvText) { const lines = csvText.split('\n').filter(line => line.trim() !== ''); if (lines.length < 2) return []; const headers = lines[0].split(',').map(h => h.replace(/^["']|["']$/g, '').trim().toUpperCase() ); return lines.slice(1).map(line => { const values = line.split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/).map(val => val ? val.replace(/^["']|["']$/g, '').trim() : '' ); const rowObj = {}; headers.forEach((header, index) => { rowObj[header] = values[index] || ''; }); return rowObj; }); } // Case-Insensitive Helper: Sheet में किसी भी मिलते-जुलते नाम से वैल्यू निकालना function getVal(row, ...possibleKeys) { if (!row) return ''; for (const key of possibleKeys) { const cleanKey = key.trim().toUpperCase(); if (row[cleanKey] !== undefined && row[cleanKey] !== '') { return row[cleanKey]; } } return ''; } // 1. Login Function (ASSOCIATE ID से Search) async function loginWithAssociateId() { const inputElement = document.getElementById('login-id') || document.getElementById('associateInput'); const associateId = inputElement ? inputElement.value.trim() : ''; if (!associateId) { alert("कृपया ASSOCIATE ID दर्ज करें!"); return; } try { // दोनों Sheets से एक साथ डेटा फ़ेच करना (No-Cache के साथ) const [res1, res2] = await Promise.all([ fetch(`${SHEET_1_URL}&_=${Date.now()}`), fetch(`${SHEET_2_URL}&_=${Date.now()}`) ]); const text1 = await res1.text(); const text2 = await res2.text(); const sheet1Records = parseCSV(text1); const sheet2Records = parseCSV(text2); // ASSOCIATE ID मैच करना (Sheet 1 और Sheet 2 दोनों में) const otData = sheet1Records.find(row => getVal(row, 'ASSOCIATE ID', 'ASSOCIATE_ID', 'EMP ID', 'ID') === associateId ); const payrollData = sheet2Records.find(row => getVal(row, 'ASSOCIATE ID', 'ASSOCIATE_ID', 'EMP ID', 'ID') === associateId ); if (!otData && !payrollData) { alert(`ASSOCIATE ID: ${associateId} का डेटा नहीं मिला!`); return; } // Login View छुपाना और Dashboard View दिखाना if (document.getElementById('login-section')) { document.getElementById('login-section').style.display = 'none'; } if (document.getElementById('dashboard-section')) { document.getElementById('dashboard-section').style.display = 'block'; } // डेटा UI पर रेंडर करना renderDashboardData(otData || {}, payrollData || {}, associateId); } catch (error) { console.error("Error loading sheets:", error); alert("डेटा लोड करने में समस्या आई। सुनिश्चित करें कि दोनों Sheets 'Anyone with link can view' पर सेट हैं।"); } } // 2. Dashboard Rendering Function function renderDashboardData(ot, payroll, id) { // User Profile Details const name = getVal(payroll, 'NAME', 'ASSOCIATE NAME') || getVal(ot, 'NAME', 'ASSOCIATE NAME') || 'Employee'; setElementText('emp-name', name); setElementText('emp-id', id); // Tab 2: Payroll & Allowances Cards (Sheet 2) setElementText('night-shift-val', '₹' + (getVal(payroll, 'NIGHT SHIFT ALLOW.', 'NIGHT SHIFT ALLOW', 'NIGHT SHIFT') || '0')); setElementText('food-allowance-val', '₹' + (getVal(payroll, 'FOOD ALLOWANCE', 'FOOD') || '0')); setElementText('active-bonus-val', '₹' + (getVal(payroll, 'ACTIVE BONUS', 'BONUS') || '0')); setElementText('b-shift-val', '₹' + (getVal(payroll, 'B SHIFT INCENTIVE', 'B SHIFT') || '0')); setElementText('att-incentive-val', '₹' + (getVal(payroll, 'ATT. INCENTIVE', 'ATT INCENTIVE', 'ATTENDANCE') || '0')); setElementText('lopr-days-val', getVal(payroll, 'LOPR DAYS', 'LOPR') || '0'); setElementText('pending-amount-val', '₹' + (getVal(payroll, 'PENDING AMOUNT', 'PENDING') || '0')); // Tab 1: OverTime Details Cards (Sheet 1) setElementText('ot-hours-val', getVal(ot, 'OT HOURS', 'OVERTIME HOURS', 'TOTAL OT') || '0'); setElementText('ot-amount-val', '₹' + (getVal(ot, 'OT AMOUNT', 'OVERTIME AMOUNT', 'TOTAL AMOUNT') || '0')); } // सुरक्षित तरीके से DOM Element का Text अपडेट करने का Helper function setElementText(id, value) { const el = document.getElementById(id); if (el) el.innerText = value; } // 3. Logout Function function logout() { const inputElement = document.getElementById('login-id') || document.getElementById('associateInput'); if (inputElement) inputElement.value = ''; if (document.getElementById('dashboard-section')) { document.getElementById('dashboard-section').style.display = 'none'; } if (document.getElementById('login-section')) { document.getElementById('login-section').style.display = 'block'; } }

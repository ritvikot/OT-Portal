// Google Sheets GViz JSON Endpoints (Data Array फॉर्मेट में) const SHEET_1_URL = 'https://docs.google.com/spreadsheets/d/1wL36ZedH-wVc7NVTiBzwPGTxYy-8fPCEnUebPmrMZQ0/gviz/tq?tqx=out:json&gid=552193494'; const SHEET_2_URL = 'https://docs.google.com/spreadsheets/d/1EZm6Kyrp6h4rUBuUv0l7HHd3IRQvqT2-10xs4YzyML0/gviz/tq?tqx=out:json&gid=0'; // Google Sheet GViz Response को 2D Array में बदलने का फंक्शन async function fetchSheetRows(url) { const res = await fetch(`${url}&_=${Date.now()}`); const text = await res.text(); // GViz JSON wrapper हटाना: /*O_o*/ google.visualization.Query.setResponse({...}); const jsonStr = text.substring(text.indexOf('{'), text.lastIndexOf('}') + 1); const data = JSON.parse(jsonStr); // Rows को सादे Array में कन्वर्ट करना return data.table.rows.map(row => row.c ? row.c.map(cell => (cell && cell.v !== null && cell.v !== undefined) ? cell.v.toString().trim() : '') : [] ); } // Login और Exact Mapping Logic async function handleLogin() { const inputId = document.getElementById('associateIdInput').value.trim(); if (!inputId) { alert("कृपया ASSOCIATE ID दर्ज करें!"); return; } try { const [sheet1Rows, sheet2Rows] = await Promise.all([ fetchSheetRows(SHEET_1_URL), fetchSheetRows(SHEET_2_URL) ]); // Sheet 2 (Payroll) में ID खोजना: Column 0 (A) = ASSOCIATE ID // अगर Sheet में ID किसी और कॉलम में है, तो index (0) बदलें const payrollRow = sheet2Rows.find(r => r[0] === inputId || r[1] === inputId); // Sheet 1 (OverTime) में ID खोजना const otRow = sheet1Rows.find(r => r[0] === inputId || r[1] === inputId); if (!payrollRow && !otRow) { alert(`ID: ${inputId} का डेटा नहीं मिला! कृपया ID चेक करें।`); return; } // View Switch document.getElementById('loginSection').style.display = 'none'; document.getElementById('dashboardSection').style.display = 'block'; // --- EXACT COLUMN MAPPING --- // (A=0, B=1, C=2, D=3, E=4, F=5, G=6, H=7, I=8...) if (payrollRow) { // Employee Name & ID document.getElementById('empName').innerText = payrollRow[1] || 'Associate'; // Column B (Name) document.getElementById('empIdDisplay').innerText = inputId; // Payroll Cards Mapping (Sheet 2 के Column Order के अनुसार इंडेक्स सेट करें) document.getElementById('nightShift').innerText = '₹' + (payrollRow[2] || '0'); // Column C: Night Shift document.getElementById('foodAllowance').innerText = '₹' + (payrollRow[3] || '0'); // Column D: Food Allowance document.getElementById('activeBonus').innerText = '₹' + (payrollRow[4] || '0'); // Column E: Active Bonus document.getElementById('bShift').innerText = '₹' + (payrollRow[5] || '0'); // Column F: B Shift Incentive document.getElementById('attIncentive').innerText = '₹' + (payrollRow[6] || '0'); // Column G: Att. Incentive document.getElementById('loprDays').innerText = payrollRow[7] || '0'; // Column H: LOPR Days document.getElementById('pendingAmount').innerText = '₹' + (payrollRow[8] || '0'); // Column I: Pending Amount } if (otRow) { // OverTime Cards Mapping (Sheet 1) document.getElementById('otHours').innerText = otRow[2] || '0'; // Column C: OT Hours document.getElementById('otAmount').innerText = '₹' + (otRow[3] || '0'); // Column D: OT Amount } } catch (err) { console.error("Data fetch error:", err); alert("डेटा फ़ेच करने में समस्या आई। Console (F12) चेक करें।"); } }
